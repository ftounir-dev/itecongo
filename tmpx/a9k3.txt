const waN_ = s => String(s || '').toLowerCase().normalize('NFD').replace(/[̀-ͯ]/g, '').replace(/[^a-z0-9]+/g, ' ').trim();
const WA_STOP = ' the what whats what s is are of for from with and our we do does how much many balance balances solde soldes du de des le la les un une au aux fournisseur fournisseurs supplier suppliers combien doit on doit owe owed give me show please stp svp merci quel quelle quels est sont sur pour par avec status statut total montant amount reste payer paid pay dette dettes stock niveau level item article ';
function waTokens_(q) { return waN_(q).split(' ').filter(w => w.length >= 3 && WA_STOP.indexOf(' ' + w + ' ') < 0); }
// Suppliers whose name matches the question (whole name in the text, or a significant word of the question inside the name).
function waFindSup_(all, q) { const n = waN_(q), T = waTokens_(q), names = {};
  all.rows.forEach(r => { if (r[13] === 'Keep' && r[2] !== 'Banks & Related') names[r[0]] = waN_(r[0]); });
  let hit = Object.keys(names).filter(k => names[k].length >= 3 && (' ' + n + ' ').indexOf(' ' + names[k] + ' ') >= 0);
  if (!hit.length) hit = Object.keys(names).filter(k => T.some(t => t.length >= 4 && names[k].split(' ').some(w => w.indexOf(t) === 0)));
  return hit; }
function waSupDetail_(all, name) { const R = all.rows.filter(r => r[0] === name && r[13] === 'Keep'), by = {};
  R.forEach(r => { const c = r[4] || 'XAF', o = by[c] || (by[c] = { rem: 0, mat: 0, inp: 0, set: 0, n: 0, items: {} });
    o.rem += r[7] || 0; o.mat += r[9] || 0; o.inp += r[10] || 0; o.set += r[12] || 0; o.n++; if (r[1]) o.items[r[1]] = 1; });
  const L = ['💳 *' + name + '*'];
  Object.keys(by).forEach(c => { const o = by[c], fx = waFx_(all, c), x = v => c === 'XAF' ? '' : ' (≈ XAF ' + waBig_(v * fx) + ')';
    L.push('Reste à payer : ' + c + ' ' + waBig_(o.rem) + x(o.rem));
    L.push('dont échu : ' + c + ' ' + waBig_(o.mat) + x(o.mat));
    if (o.inp) L.push('Paiements en cours : ' + c + ' ' + waBig_(Math.abs(o.inp)) + x(Math.abs(o.inp)));
    if (o.set && Math.abs(o.set - o.rem) > 1) L.push('À régler : ' + c + ' ' + waBig_(o.set) + x(o.set));
    L.push('Lignes : ' + o.n + ' · ' + Object.keys(o.items).slice(0, 4).join(', ')); });
  return L.join('\n'); }
function waFindItems_(all, q) { const n = waN_(q), T = waTokens_(q), it = ((all.stock || {}).items || []);
  let hit = it.filter(x => { const k = waN_(x.name); return k.length >= 3 && n.indexOf(k) >= 0; });
  if (!hit.length) hit = it.filter(x => { const W = waN_(x.name).split(' '); return T.some(t => t.length >= 4 && W.some(w => w.indexOf(t) === 0)); });
  return hit.slice(0, 6); }

function waAnswer_(u, raw) { const acc = u.access || [], has = k => acc.indexOf(k) >= 0, no = '🔒 Cette section n’est pas ouverte pour votre profil.';
  const q = waN_(raw);
  if (!q || /^(aide|help|menu|bonjour|bonsoir|hi|hello|salut|start|hey)( |$)/.test(q)) return 'Bonjour ' + u.name + ' 👋\n\n' + WA_MENU;
  const all = allData_(), w1 = q.split(' ')[0], short = q.split(' ').length <= 2;
  const cmd = k => short && k.test(w1);
  if (cmd(/^(point|resume|summary|rapport|recap)$/)) return waSummaryText_(all, acc);
  if (cmd(/^(banques?|banks?|treso|tresorerie)$/)) return has('banks') ? waBanksDetail_(all) : no;
  if (cmd(/^(fournisseurs?|suppliers?|echus?|ap)$/)) return has('suppliers') ? waSupText_(all, 10) : no;
  if (cmd(/^(stock|stocks|mp)$/)) return has('stock') ? waStockLowText_(all) : no;
  if (cmd(/^(eta|etas|arrivees?|arrivals?)$/)) { const E = waEtaList_(all, 14, acc); return E === null ? no : E; }
  if (cmd(/^(cuves?|gasoil|fuel|tanks?)$/)) return has('fleet') ? waFuelText_(all) : no;
  // A named supplier or item anywhere in the question
  if (has('suppliers')) { const S = waFindSup_(all, raw); if (S.length) return S.slice(0, 3).map(s => waSupDetail_(all, s)).join('\n\n'); }
  if (has('stock')) { const it = waFindItems_(all, raw);
    if (it.length) return it.map(x => '📦 *' + x.name + '*' + (x.fam ? ' (' + x.fam + ')' : '') + ' : ' + waMo_(x.act) + (x.eta ? ' · prochaine arrivée ' + x.eta : '')).join('\n'); }
  // Topic words anywhere in the question
  if (/banq|bank|treso|decouvert|overdraft|cheque|cash/.test(q)) return has('banks') ? waBanksDetail_(all) : no;
  if (/fournisseur|supplier|echu|overdue|dette|payable|owe/.test(q)) return has('suppliers') ? waSupText_(all, 10) : no;
  if (/stock|rupture|matiere|raw material/.test(q)) return has('stock') ? waStockLowText_(all) : no;
  if (/eta|arriv|livraison|shipment|bateau|conteneur|container|delivery/.test(q)) { const E = waEtaList_(all, 14, acc); return E === null ? no : E; }
  if (/cuve|gasoil|fuel|tank|carburant|diesel/.test(q)) return has('fleet') ? waFuelText_(all) : no;
  if (/point|resume|summary|rapport|situation/.test(q)) return waSummaryText_(all, acc);
  return 'Je n’ai pas compris « ' + String(raw).slice(0, 40) + ' ».\n\n' + WA_MENU; }
function waStockLowText_(all) { const S = waStockLow_(all);
  return ['📦 *Stock < 1 mois* : ' + (S.length ? S.length + ' article(s)' : 'aucun')].concat(S.map(x => '• ' + x.name + ' : ' + waMo_(x.act))).join('\n'); }
