function waAlive_(u, raw, ctx, all, acc) { const q = waN_(raw), Q = ' ' + q + ' ', has = k => acc.indexOf(k) >= 0, no = T_('🔒 Cette section n’est pas ouverte pour votre profil.', '🔒 This section is not open for your profile.');

  if (pend && /^(cancel|annuler|non|no|stop)$/.test(q)) { CacheService.getScriptCache().remove('wp_' + u.u); return T_('❌ Annulé.', '❌ Cancelled.'); }
  if (/^(aide|help|exemples?|examples?)$/.test(q)) return waHelp_();
  if (/^(rappelle|rappel|remind|reminder)/.test(q)) return / (mes rappels|my reminders|rappels|reminders) /.test(Q) && q.split(' ').length <= 3 ? waReminderList_(u) : waReminder_(u, raw, WA_FROM || waNum_(u.phone));
  if (/ (update|maj|mettre a jour|mets a jour|modifier|change|changer|set|nouvelle) /.test(Q) && / (eta|etas) /.test(Q) && /\d{1,2}[\/.-]\d{1,2}/.test(raw)) { if (!has('projects')) return no; const r = waEtaUpdate_(u, raw, all); if (r) return r; }
  if (/ (pdf|document|fichier|file) /.test(Q)) {
    if (/ (factures?|invoices?|aging|ageing|echeancier) /.test(Q)) { if (!has('suppliers')) return no; const p = waAgingPdf_(raw); if (p) { WA_DOC = p.blob; return '📄 ' + T_('Factures ', 'Invoices ') + p.sup + ' · ' + p.n + T_(' ouvertes', ' open'); } }
    if (/ (stock|stocks|matieres?|mp) /.test(Q)) { if (!has('stock')) return no; WA_DOC = waStockPdf_(all); return '📄 ' + T_('Stock matières premières', 'Raw materials stock'); }
    if (WA_SALES_W.test(Q)) { if (!has('stock')) return no; const b = waSalesPdf_(raw); if (b) { WA_DOC = b; return '📄 ' + b.getName().replace(/\.pdf$/, ''); } } }
  if (/ (rupture|ruptures|runout|run out|run outs) /.test(Q)) { if (!has('stock')) return no; return waRunoutText_(all) || T_('✅ Aucun risque de rupture à 2 mois.', '✅ No run-out risk within 2 months.'); }
  return null; }
