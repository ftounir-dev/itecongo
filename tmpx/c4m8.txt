function waIncoming_(p) { const m = p && p.message; if (!m || !m.id) return;
  if (m.kapso && m.kapso.direction && m.kapso.direction !== 'inbound') return;
  const c = waCfg_(); if (p.phone_number_id && String(p.phone_number_id) !== String(c.phone)) return;
  const cache = CacheService.getScriptCache(), lock = LockService.getScriptLock(); lock.waitLock(20000);
  try { if (cache.get('wm_' + m.id)) return; cache.put('wm_' + m.id, '1', 21600); } finally { lock.releaseLock(); }
  const from = waNum_(m.from || (p.conversation || {}).phone_number); if (!from) return;
  const u = waUserByPhone_(from);
  const text = (m.text && m.text.body) || (m.kapso && m.kapso.content) || '';
  WA_L = waLang_(text);
  if (!u) { if (!cache.get('wx_' + from)) { cache.put('wx_' + from, '1', 21600); waSendTo_(from, T_('Ce numéro n’est pas autorisé pour l’agent ITE Congo.', 'This number is not authorised for the ITE Congo agent.')); } console.log('WA unknown ' + from); return; }

  let ans; try { ans = waAnswer_(u, text); } catch (err) { ans = T_('⚠️ Erreur, réessayez dans un instant.', '⚠️ Error, please try again in a moment.'); console.error('waAnswer ' + err); }
  console.log('WA ' + u.u + ' ' + WA_L + ' "' + String(text).slice(0, 60) + '" -> ' + waSendTo_(from, ans)); }
