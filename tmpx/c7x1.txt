// Conversation memory (15 min per person): what the last answer was about, so "any close ETA?" follows on from it.
var WA_NEWCTX = {};
function waCtxGet_(k) { try { return JSON.parse(CacheService.getScriptCache().get('wc_' + k) || '{}'); } catch (e) { return {}; } }
function waCtxSet_(k, o) { o.t = Date.now(); CacheService.getScriptCache().put('wc_' + k, JSON.stringify(o), 900); }
function waItemEtas_(all, names) { const t = waToday_(), it = ((all.stock || {}).items || []), L = [];
  names.slice(0, 6).forEach(n => { const x = it.find(i => i.name === n); if (!x) return;
    const A = (x.bl || []).filter(b => b.eta).map(b => ({ d: waPd_(b.eta), qty: b.qty, mo: b.add })).filter(b => b.d && b.d >= t);
    if (!A.length && x.eta) { const d = waPd_(x.eta); if (d && d >= t) A.push({ d: d, qty: null, mo: x.add }); }
    A.sort((a, b) => a.d - b.d);
    L.push('🚢 *' + n + '* · ' + T_('stock ', 'stock ') + waMo_(x.act));
    if (!A.length) L.push(T_('• Aucune arrivée prévue', '• No arrival planned'));
    A.forEach(b => L.push('• ' + Utilities.formatDate(b.d, Session.getScriptTimeZone(), 'dd/MM/yyyy') + waQty_({ qty: b.qty, mo: b.mo, uom: x.uom }))); });
  return L.length ? L.join('\n') : null; }
function waFollowUp_(all, q, ctx, has) { if (!ctx || !ctx.t || Date.now() - ctx.t > 900000) return null;
  if (/(^| )(all|tous|toutes|tout|global|general|everything)( |$)/.test(q)) return null;
  const etaW = /(^| )(eta|etas|etd|arriv\w*|livr\w*|deliver\w*|ship\w*|quand|when|next|prochaine?s?|close|soon|bientot|coming|upcoming)( |$)/.test(q);
  const stW = /(^| )(stock|stocks|niveau|level|cover|couverture|mois|months|left|reste)( |$)/.test(q);
  const balW = /(^| )(solde|balance|owe|doit|dette|outstanding|payer|pay|overdue|echu|montant|amount)( |$)/.test(q);
  const tail = T_('\n\n_(« eta tout » pour toutes les arrivées)_', '\n\n_(“eta all” for every arrival)_');
  if (etaW && ctx.items && ctx.items.length && has('stock')) { WA_NEWCTX.items = ctx.items; const r = waItemEtas_(all, ctx.items); if (r) return r + tail; }
  const S = (ctx.sups || []).concat(ctx.ppm || []);
  if (etaW && S.length && has('projects')) { const R = (all.ppm || []).filter(r => r[13] !== 'Hide' && S.indexOf(r[0]) >= 0); WA_NEWCTX.ppm = S;
    if (R.length) return R.slice(0, 5).map(waPPMDetail_).join('\n\n') + tail; }
  if (stW && ctx.items && ctx.items.length && has('stock')) { WA_NEWCTX.items = ctx.items; const it = ((all.stock || {}).items || []).filter(x => ctx.items.indexOf(x.name) >= 0);
    if (it.length) return it.map(x => '📦 *' + x.name + '*' + (x.fam ? ' (' + x.fam + ')' : '') + ' : ' + waMo_(x.act)).join('\n'); }
  if (balW && S.length && has('suppliers')) { const N = S.filter(s => all.rows.some(r => r[0] === s)); WA_NEWCTX.sups = N;
    if (N.length) return N.slice(0, 3).map(s => waSupDetail_(all, s)).join('\n\n'); }
  return null; }
