// ===== Master Plan PDF drawn like the app's "Save as PDF" (same rows, colours, markers), built in Google Slides then exported =====
function waStage_(r, t) { const note = String(r[14] || ''), R = (r[15] || []).map(waPd_).filter(Boolean), E = (r[16] || []).map(waPd_).filter(Boolean), A = (r[17] || []).map(waPd_).filter(Boolean);
  const arr = /arriv/i.test((r[17] || []).join(' ')), eta = A.length ? new Date(Math.max.apply(null, A)) : null, etd = E.length ? new Date(Math.min.apply(null, E)) : null;
  if (/(^|\W)ite(\W|$)/i.test(note)) return 100; if ((eta && eta <= t) || arr) return 100;
  if ((etd && etd <= t) || (eta && !etd)) { const s0 = etd || new Date(+eta - 60 * 864e5), f = eta ? Math.min(1, Math.max(0, (t - s0) / (eta - s0))) : 0; return Math.round(30 + 70 * f); }
  if (E.length || R.some(d => d <= t)) return 30; if (R.length) return 15; return 0; }
function waMPData_(all, famPick) { const t = waToday_(), G = {}, order = [];
  (all.ppm || []).forEach(r => { if (r[2] !== 'Projects' || r[13] === 'Hide' || /mohamad ?moussa/i.test(r[0])) return;
    const fam = String(r[3] || '').trim() || 'Other'; if (famPick && famPick.length && famPick.indexOf(fam) < 0) return;
    const k = fam + '|' + String(r[1]).trim().toLowerCase().replace(/\s+/g, ' ') + '|' + r[0]; if (!G[k]) { G[k] = { fam: fam, sup: r[0], name: String(r[1]).trim().replace(/\s+/g, ' '), R: [], E: [], A: [], M: {}, w: 0, wd: 0, lo: 100 }; order.push(k); }
    const g = G[k], dp = waStage_(r, t), w = Math.abs((r[5] || 0) * waFx_(all, r[4])) || 1;
    (r[15] || []).map(waPd_).filter(Boolean).forEach(d => g.R.push(d)); (r[16] || []).map(waPd_).filter(Boolean).forEach(d => g.E.push(d));
    (r[17] || []).forEach((s, i) => { const d = waPd_(s); if (d) { g.A.push(d); if ((r[18] || [])[i]) g.M[+d] = 1; } });
    g.w += w; g.wd += w * dp; g.lo = Math.min(g.lo, dp); });
  const mn = a => a.length ? new Date(Math.min.apply(null, a)) : null, mx = a => a.length ? new Date(Math.max.apply(null, a)) : null;
  return order.map(k => { const g = G[k]; return { fam: g.fam, sup: g.sup, name: g.name, pct: g.w ? g.wd / g.w : 0, done: g.lo >= 100, r: mn(g.R), e: mn(g.E), a: mx(g.A),
    A: [...new Set(g.A.map(Number))].map(x => new Date(x)), M: g.M }; }); }

function waMasterPlanPdf_(all, raw) { const q = waN_(raw), fams0 = [...new Set((all.ppm || []).map(r => String(r[3] || '').trim()).filter(String))], pick = fams0.filter(f => (' ' + q + ' ').indexOf(' ' + waN_(f) + ' ') >= 0);
  const rows = waMPData_(all, pick), now = waToday_(), day = 864e5, tz = Session.getScriptTimeZone(), FR = WA_L !== 'en';
  const dated = rows.filter(x => x.r || x.e || x.a), nod = rows.filter(x => !(x.r || x.e || x.a)), undated = nod.filter(x => !x.done), atite = nod.filter(x => x.done);
  const ds = dated.reduce((a, x) => a.concat([x.r, x.e, x.a].filter(Boolean)), [now]);
  let s = new Date(Math.min.apply(null, ds) - 14 * day); s = new Date(s.getFullYear(), s.getMonth(), 1);
  let e = new Date(Math.max.apply(null, ds) + 14 * day); e = new Date(e.getFullYear(), e.getMonth() + 1, 1);
  const F = {}; dated.forEach(x => (F[x.fam] = F[x.fam] || []).push(x)); atite.forEach(x => (F[x.fam] = F[x.fam] || []).push(Object.assign({ ite: true }, x)));
  const key = x => x.ite ? 9e15 : +(x.a || x.e || x.r), fams = Object.keys(F).sort((a, b) => Math.min.apply(null, F[a].map(key)) - Math.min.apply(null, F[b].map(key)));
  const MO = FR ? ['janv', 'févr', 'mars', 'avr', 'mai', 'juin', 'juil', 'août', 'sept', 'oct', 'nov', 'déc'] : ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
  const dd = d => Utilities.formatDate(d, tz, 'dd/MM'), dsT = Utilities.formatDate(new Date(), tz, 'dd/MM/yyyy');
  // Page geometry (points, 16:9 slide 720 x 405)
  const PW = 720, PH = 405, ML = 18, LW = 150, VX = 676, TX0 = ML + LW + 6, TX1 = VX - 6, TW = TX1 - TX0, RH = 19, FH = 18, TOP = 92, BOT = PH - 14;
  const X = d => TX0 + (d - s) / (e - s) * TW;
  const pres = SlidesApp.create('tmp Master Plan ' + Date.now()), id = pres.getId();
  try {
    pres.getSlides()[0].remove();
    const C = { ink: '#14231F', soft: '#5E6E68', line: '#DDE4E0', grn: '#2F8A72', pr: '#B8C4CC', wt: '#A9CBC1', tr: '#2F8A72', dn: '#ACD0C6', mk: '#4A6278', a: '#1E5D4C', am: '#1F5FA8', now: '#C0563B' };
    const txt = (sl, t, x, y, w, h, o) => { o = o || {}; const b = sl.insertTextBox(t, x - 3, y - 3, w + 6, h + 6), st = b.getText().getTextStyle();
      st.setFontFamily('Inter').setFontSize(o.size || 7).setBold(!!o.bold).setForegroundColor(o.color || C.ink);
      b.getText().getParagraphStyle().setParagraphAlignment(o.align === 'r' ? SlidesApp.ParagraphAlignment.END : o.align === 'c' ? SlidesApp.ParagraphAlignment.CENTER : SlidesApp.ParagraphAlignment.START);
      b.setContentAlignment(SlidesApp.ContentAlignment.MIDDLE); if (o.fill) b.getFill().setSolidFill(o.fill); return b; };
    const rect = (sl, type, x, y, w, h, fill, border) => { const r = sl.insertShape(type, x, y, w, h); r.getFill().setSolidFill(fill);
      if (border) { r.getBorder().setWeight(border[1]).getLineFill().setSolidFill(border[0]); } else r.getBorder().setTransparent(); return r; };
    const line = (sl, x1, y1, x2, y2, col, wt, dash) => { const l = sl.insertLine(SlidesApp.LineCategory.STRAIGHT, x1, y1, x2, y2); l.getLineFill().setSolidFill(col); l.setWeight(wt); if (dash) l.setDashStyle(SlidesApp.DashStyle.DASH); return l; };
    const months = []; for (let m = new Date(s); m < e; m = new Date(m.getFullYear(), m.getMonth() + 1, 1)) months.push(m);
    const nRows = dated.length + atite.length; let sl, y, page = 0;
    const newPage = () => { sl = pres.appendSlide(SlidesApp.PredefinedLayout.BLANK); sl.getBackground().setSolidFill('#FFFFFF'); page++;
      rect(sl, SlidesApp.ShapeType.RECTANGLE, 0, 0, PW, 40, '#13231E'); rect(sl, SlidesApp.ShapeType.RECTANGLE, 0, 40, PW, 2.5, C.grn);
      const h = sl.insertTextBox('ITE Congo  | Business Platform   Master Plan', ML - 3, 8, 420, 26), tr = h.getText(); tr.getTextStyle().setFontFamily('Inter').setFontSize(14).setBold(true).setForegroundColor('#FFFFFF');
      tr.getRange(10, 30).getTextStyle().setBold(false).setFontSize(11).setForegroundColor('#B9CBC4'); tr.getRange(30, tr.asString().length - 1).getTextStyle().setForegroundColor('#7FD0B3');
      txt(sl, dsT + ' · ' + dated.length + (FR ? ' projets datés' : ' projects with dates') + (page > 1 ? ' · p.' + page : ''), PW - 260 - ML, 12, 260, 16, { size: 8.5, color: '#B9CBC4', align: 'r' });
      // legend
      let lx = ML; const ly = 50; const lg = (draw, label) => { draw(lx, ly); const b = txt(sl, label, lx + 18, ly - 4, 110, 10, { size: 6.5, color: C.soft }); lx += 22 + label.length * 3.3; };
      if (page === 1) {
        lg((x, yy) => rect(sl, SlidesApp.ShapeType.ROUND_RECTANGLE, x, yy - 2, 14, 4.5, C.pr), FR ? 'En fabrication' : 'In production');
        lg((x, yy) => rect(sl, SlidesApp.ShapeType.ROUND_RECTANGLE, x, yy - 2, 14, 4.5, C.wt), FR ? 'Prêt, en attente d’expédition' : 'Ready, awaiting shipment');
        lg((x, yy) => rect(sl, SlidesApp.ShapeType.ROUND_RECTANGLE, x, yy - 2, 14, 4.5, C.tr), FR ? 'Acheminement vers ITE' : 'Shipping to ITE');
        lg((x, yy) => rect(sl, SlidesApp.ShapeType.DIAMOND, x + 4, yy - 3, 6, 6, '#FFFFFF', [C.mk, 1]), FR ? 'Disponibilité' : 'Readiness');
        lg((x, yy) => rect(sl, SlidesApp.ShapeType.RECTANGLE, x + 4, yy - 2.5, 5, 5, C.mk), 'ETD');
        lg((x, yy) => rect(sl, SlidesApp.ShapeType.ELLIPSE, x + 4, yy - 3, 6.5, 6.5, C.a, ['#FFFFFF', 1]), 'ETA');

      const hy = 66; months.forEach(m => { const x = X(m); line(sl, x, hy, x, BOT, C.line, 0.5);

      line(sl, ML, hy + 16, PW - ML, hy + 16, C.line, 0.6);
      line(sl, X(now), hy, X(now), BOT, C.now, 1.2); txt(sl, FR ? 'Aujourd’hui' : 'Today', X(now) + 2, hy - 9, 50, 8, { size: 6, color: C.now, bold: true });

    newPage();
    fams.forEach(fm => { const L = F[fm].sort((a, b) => key(a) - key(b));

      ft.setContentAlignment(SlidesApp.ContentAlignment.MIDDLE); line(sl, ML, y + FH, PW - ML, y + FH, C.line, 0.5); y += FH;
      L.forEach(x => { if (y + RH > BOT) newPage(); const cy = y + RH / 2;
        const nb = sl.insertTextBox(x.sup + '\n' + x.name, ML - 3, y - 2, LW + 6, RH + 4), nt = nb.getText(); nb.setContentAlignment(SlidesApp.ContentAlignment.MIDDLE);
        nt.getTextStyle().setFontFamily('Inter').setFontSize(6).setForegroundColor(C.soft); nt.getRange(0, x.sup.length).getTextStyle().setBold(true).setFontSize(7).setForegroundColor(C.ink);
        nt.getParagraphStyle().setSpaceAbove(0).setSpaceBelow(0).setLineSpacing(95);
