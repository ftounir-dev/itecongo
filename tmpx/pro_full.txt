// Stock PDF layout: produces drawing primitives (pt units, 720x405 slide). Shared by Apps Script (Slides) and the local preview.
function waStockLayout_(items, opt) {
  opt = opt || {}; const fr = opt.lang !== 'en', T = (a, b) => fr ? a : b, W = 720, H = 405, P = [], pages = [];
  const C = { dark: '#13231E', green: '#2F8A72', mint: '#7FD0B3', mute: '#5E6E68', line: '#DDE4E0', band: '#E3F1EB', bandTx: '#1F6E58', zebra: '#F6FAF8', track: '#E8EEEB', proj: '#A9C9E8', red: '#C0392B', orange: '#E67E22', amber: '#D4A017', ok: '#2F8A72', ink: '#14231F' };
  const col = a => a == null ? C.mute : a <= 0.05 ? C.red : a < 1 ? C.orange : a < 3 ? C.amber : C.ok;
  const nb = v => (Math.round(v * 10) / 10).toFixed(1), nf = v => Math.round(v).toLocaleString('en-US');
  const MAXM = 6, X = { name: 16, stock: 214, bar: 300, barW: 150, act: 452, eta: 494, qty: 548, proj: 612, end: 704 };
  let page, y;
  const R = (x, y, w, h, fill, rnd) => page.push({ t: 'r', x, y, w, h, fill, rnd: !!rnd });
  const Tx = (x, y, w, h, s, o) => page.push(Object.assign({ t: 'x', x, y, w, h, s: String(s), size: 7.5, color: C.ink, bold: false, al: 'L' }, o || {}));
  const header = (first) => {
    R(0, 0, W, 38, C.dark); R(0, 38, W, 2.5, C.green);
    Tx(14, 6, 260, 26, 'ITE Congo', { size: 15, bold: true, color: '#FFFFFF' });
    Tx(104, 9, 160, 22, '| Business Platform', { size: 10, color: '#9FB5AD' });
    Tx(330, 4, 376, 18, T('Stock matières premières', 'Raw materials stock'), { size: 12, bold: true, color: C.mint, al: 'R' });
    Tx(330, 20, 376, 14, T('Mis à jour ', 'Updated ') + (opt.asOf || ''), { size: 8, color: '#9FB5AD', al: 'R' });
  };
  const colHead = (yy) => {
    R(X.name - 4, yy, X.end - X.name + 8, 15, C.dark);
    const h = { color: '#FFFFFF', size: 7, bold: true };
    Tx(X.name, yy, 190, 15, T('Article', 'Item'), h); Tx(X.stock, yy, 80, 15, 'Stock', Object.assign({ al: 'R' }, h));
    Tx(X.bar, yy, X.barW + 40, 15, T('Couverture actuelle (0–6 mois)', 'Current cover (0–6 months)'), h);
    Tx(X.eta, yy, 52, 15, T('Proch. ETA', 'Next ETA'), h); Tx(X.qty, yy, 62, 15, T('Qté en route', 'Qty on way'), Object.assign({ al: 'R' }, h));
    Tx(X.proj, yy, 92, 15, T('Après arrivages', 'After arrivals'), Object.assign({ al: 'R' }, h));
    return yy + 15;
  };
  const newPage = (first) => { page = []; pages.push(page); header(first); y = first ? 112 : 50; y = colHead(y); };
  // KPIs on page 1
  newPage(true);
  const n0 = items.filter(x => x.act != null && x.act <= 0.05).length, n1 = items.filter(x => x.act != null && x.act > 0.05 && x.act < 1).length,
    n3 = items.filter(x => x.act != null && x.act >= 1 && x.act < 3).length, nOk = items.filter(x => x.act != null && x.act >= 3).length;
  [[n0, T('En rupture', 'Out of stock'), C.red], [n1, T('Moins d’1 mois', 'Under 1 month'), C.orange], [n3, T('1 à 3 mois', '1 to 3 months'), C.amber], [nOk, T('Plus de 3 mois', 'Over 3 months'), C.ok]].forEach((k, i) => {
    const x = 12 + i * 176.5; R(x, 50, 168, 50, '#F6FAF8'); R(x, 50, 4, 50, k[2]);
    Tx(x + 10, 52, 150, 28, k[0], { size: 20, bold: true, color: k[2] }); Tx(x + 10, 80, 150, 16, k[1] + ' · ' + T('articles', 'items'), { size: 8, color: C.mute });
  });
  const RH = 15, BOT = 380;
  const fams = []; items.forEach(x => { if (fams.indexOf(x.fam) < 0) fams.push(x.fam); });
  fams.forEach(f => {
    const L = items.filter(x => x.fam === f);
    if (y + RH * 3 > BOT) newPage(false);
    R(X.name - 4, y, X.end - X.name + 8, RH + 1, C.band); R(X.name - 4, y + RH, X.end - X.name + 8, 1, C.green);
    Tx(X.name, y, 400, RH, f.toUpperCase() + '  ·  ' + L.length + T(' articles', ' items'), { size: 8, bold: true, color: C.bandTx }); y += RH + 1;
    L.forEach((x, i) => {
      if (y + RH > BOT) { newPage(false); Tx(X.name, y, 400, RH, f.toUpperCase() + T(' (suite)', ' (cont.)'), { size: 7.5, bold: true, color: C.bandTx }); y += RH; }
      if (i % 2) R(X.name - 4, y, X.end - X.name + 8, RH, C.zebra);
      Tx(X.name, y, 196, RH, x.name, { size: 7.5, bold: true });
      Tx(X.stock, y, 80, RH, (x.stock != null ? nf(x.stock) : '–') + (x.uom ? ' ' + x.uom : ''), { al: 'R', color: C.mute });
      const bw = X.barW, by = y + 4.5, bh = RH - 9; R(X.bar, by, bw, bh, C.track, true);
      const pj = x.proj != null && x.proj > (x.act || 0) ? Math.min(x.proj, MAXM) : null;
      if (pj) R(X.bar, by, bw * pj / MAXM, bh, C.proj, true);
      if (x.act != null && x.act > 0.05) R(X.bar, by, Math.max(3, bw * Math.min(x.act, MAXM) / MAXM), bh, col(x.act), true);
      Tx(X.act, y, 44, RH, x.act == null ? '–' : x.act <= 0.05 ? T('Rupture', 'Out') : nb(x.act) + T(' m', ' mo'), { bold: true, color: col(x.act) });
      const A = (x.bl || []).filter(b => b.eta).map(b => b.eta); const eta = A[0] || x.eta || '';
      Tx(X.eta, y, 56, RH, eta ? String(eta).slice(0, 5) : '–', { color: eta ? '#2F6FA8' : C.mute, bold: !!eta });
      const q = (x.bl || []).reduce((s, b) => s + (b.qty || 0), 0);
      Tx(X.qty, y, 62, RH, q ? nf(q) : '–', { al: 'R', color: q ? C.ink : C.mute });
      Tx(X.proj, y, 92, RH, pj ? '→ ' + nb(x.proj) + T(' mois', ' months') : '', { al: 'R', bold: true, color: '#2F6FA8' });
      y += RH;
    });
    y += 4;
  });
  // footer + legend on every page
  pages.forEach((pg, i) => { page = pg;
    R(0, 388, W, 0.8, C.line);
    const lg = [[C.red, T('Rupture', 'Out')], [C.orange, T('< 1 mois', '< 1 month')], [C.amber, T('1–3 mois', '1–3 months')], [C.ok, T('> 3 mois', '> 3 months')], [C.proj, T('Après arrivages prévus', 'After planned arrivals')]];
    let lx = 16; lg.forEach(l => { R(lx, 393, 9, 6, l[0], true); Tx(lx + 11, 389, 120, 14, l[1], { size: 6.5, color: C.mute }); lx += l[1].length * 3.4 + 30; });
    Tx(560, 389, 146, 14, T('Page ', 'Page ') + (i + 1) + ' / ' + pages.length, { size: 6.5, color: C.mute, al: 'R' });
  });
  return { W, H, pages };
}
