// ===== Smart search over every tab of the A/P and Stock spreadsheets (no AI) =====
// Users, Messages, PPM_Store and backup tabs are never read. Each tab follows the section rights of the app.
const WA_SRC = [{ id: SPREADSHEET_ID, sec: n => /^a\/p$|^aging/i.test(n) ? 'suppliers' : /bank/i.test(n) ? 'banks' : /^ppm$/i.test(n) ? 'projects' : null },
  { id: STOCK_ID, sec: n => 'stock' }];
const WA_SKIP = /^(users|messages|ppm_store)$|backup/i;
function waSheetsRefresh_() { const out = [];
  WA_SRC.forEach(src => { SpreadsheetApp.openById(src.id).getSheets().forEach(sh => { const name = sh.getName(), sec = src.sec(name);
    if (WA_SKIP.test(name) || !sec || sh.isSheetHidden()) return;
    const v = sh.getDataRange().getDisplayValues().map(r => { r = r.map(x => String(x).replace(/\s+/g, ' ').trim().slice(0, 60)); while (r.length && !r[r.length - 1]) r.pop(); return r; });
    const cnt = r => r.filter(String).length, isNum = x => /^[\s$€(-]*[\d.,]+\s*%?\)?\s*\w{0,4}$/.test(x);
    let h = v.findIndex(r => cnt(r) >= 3); if (h < 0) return;
    const H1 = v[h].slice(), nx = v[h + 1] || [], two = cnt(nx) >= 3 && nx.filter(x => x && isNum(x)).length <= cnt(nx) / 2;
    for (let i = 1; i < H1.length; i++) if (!H1[i] && two && nx[i]) H1[i] = H1[i - 1]; // months spanning several columns
    const W = Math.max(H1.length, nx.length), H = []; for (let i = 0; i < W; i++) H.push([H1[i] || '', two ? nx[i] || '' : ''].filter(String).join(' '));
    const rows = []; v.slice(h + (two ? 2 : 1)).forEach(r => { if (cnt(r) >= 2) rows.push(r); });
    out.push({ n: name, s: sec, h: H, r: rows }); }); });
  const s = JSON.stringify(out), c = CacheService.getScriptCache(), o = {}; let k = 0;
  for (let i = 0; i < s.length; i += 90000) o['ws_' + (k++)] = s.slice(i, i + 90000);
  o.ws_n = String(k); c.putAll(o, 1500); return out; }
function waSheets_() { const c = CacheService.getScriptCache(), n = Number(c.get('ws_n') || 0);
  if (n) { const K = []; for (let i = 0; i < n; i++) K.push('ws_' + i); const P = c.getAll(K); let s = '';
    for (let i = 0; i < n; i++) { if (P['ws_' + i] == null) { s = ''; break; } s += P['ws_' + i]; } if (s) try { return JSON.parse(s); } catch (e) {} }
  return waSheetsRefresh_(); }

// Words that point to a tab.
const WA_HINT = [[/^(forecast|prevision|previsions|objectif|objectifs|target|targets|achievement|realisation)/, /sales forecast/i],
  [/^(nomenclature|recette|recipe|bom|composition|formule|formula)/, /nomenclature/i], [/^(needs|besoin|besoins|historique|history)/, /sales & fp needs/i],
  [/^(aging|ageing|facture|factures|invoice|invoices|echeance|maturity)/, /^aging/i], [/^(ppm)/, /^ppm$/i], [/^(raw|matieres?)$/, /raw materials/i]];
function waSearch_(raw, acc, force) { const has = k => acc.indexOf(k) >= 0, S = waSheets_().filter(t => has(t.s));
  let T = waTokens_(raw).map(t => WA_SYN[t] || t), only = null;
  T = T.filter(t => { const m = WA_HINT.find(x => x[0].test(t)); if (m) { only = only || []; only.push(m[1]); return false; } return true; });
  const tabs = only ? S.filter(t => only.some(re => re.test(t.n))) : S;
  if (!tabs.length || (!T.length && !only)) return null; if (!force && !only) return null;
  const hits = [];
  tabs.forEach(t => t.r.forEach(r => { const W = waN_(r.join(' ')).split(' ');
    const sc = T.filter(x => W.some(w => w.indexOf(x) === 0)).length; if (sc || !T.length) hits.push({ t: t, r: r, sc: sc }); }));
  if (!hits.length) return null;
  const best = Math.max.apply(null, hits.map(x => x.sc)), top = hits.filter(x => x.sc === best).slice(0, T.length ? 4 : 6);
  const more = hits.filter(x => x.sc === best).length - top.length;
  const out = top.map(x => { const r = x.r, h = x.t.h, P = [];
    r.forEach((v, i) => { if (v) P.push([h[i] || '', v]); });
    const id = P.slice(0, 3).map(p => p[1]).join(' · '), rest = P.slice(3), show = rest.length > 14 ? rest.slice(-12) : rest;
    return '📄 *' + x.t.n + '* · ' + id + (show.length ? '\n' + show.map(p => '• ' + (p[0] ? p[0] + ' : ' : '') + p[1]).join('\n') : ''); });
  return out.join('\n\n') + (more > 0 ? T_('\n\n… et ' + more + ' autre(s) ligne(s). Précisez votre recherche.', '\n\n… and ' + more + ' more row(s). Refine your search.') : ''); }
