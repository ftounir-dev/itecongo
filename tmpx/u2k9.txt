const WA_MON = { jan: 0, janv: 0, janvier: 0, january: 0, feb: 1, fev: 1, fevr: 1, fevrier: 1, february: 1, mar: 2, mars: 2, march: 2, apr: 3, avr: 3, avril: 3, april: 3, may: 4, mai: 4, jun: 5, juin: 5, june: 5, jul: 6, juil: 6, juillet: 6, july: 6, aug: 7, aou: 7, aout: 7, august: 7, sep: 8, sept: 8, septembre: 8, september: 8, oct: 9, octobre: 9, october: 9, nov: 10, novembre: 10, november: 10, dec: 11, decembre: 11, december: 11 };
const WA_MLAB = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
// Months named in the question, as sheet labels ("octobre 2026" / "oct 26" / "10/2026" -> "Oct-2026").
function waMonths_(raw) { const W = waN_(raw).split(' '), out = [], y0 = new Date().getFullYear();
  for (let i = 0; i < W.length; i++) { const m = WA_MON[W[i]]; if (m == null) continue;
    let y = /^\d{4}$/.test(W[i + 1] || '') ? +W[i + 1] : /^\d{2}$/.test(W[i + 1] || '') ? 2000 + +W[i + 1] : y0;
    out.push(WA_MLAB[m] + '-' + y); }
  String(raw).replace(/\b(\d{1,2})[\/-](20\d{2})\b/g, (a, m, y) => { if (+m >= 1 && +m <= 12) out.push(WA_MLAB[+m - 1] + '-' + y); return a; });
  return out; }
const waNum2_ = s => { const v = parseFloat(String(s || '').replace(/[,\s]/g, '').replace(/^\((.*)\)$/, '-$1')); return isNaN(v) ? 0 : v; };
const WA_SALES_W = / (sales|sale|ventes?|vendu|vendus|sold|selling|chiffres?|turnover|volumes?) /;
function waSales_(raw, acc) { const q = ' ' + waN_(raw) + ' '; if (!WA_SALES_W.test(q) || acc.indexOf('stock') < 0) return null;
  const t = waSheets_().find(x => /sales & fp needs/i.test(x.n)); if (!t) return T_('📊 Onglet Sales & FP Needs introuvable.', '📊 Sales & FP Needs tab not found.');
  const H = t.h.map(h => String(h).trim()), mIdx = lab => H.findIndex(h => h.toLowerCase() === lab.toLowerCase());
  const isM = h => /^[A-Z][a-z]{2}-\d{4}$/.test(h);
  let M = waMonths_(raw).map(mIdx).filter(i => i >= 0);
  if (!M.length) { for (let i = H.length - 1; i >= 0; i--) if (isM(H[i]) && t.r.some(r => waNum2_(r[i]))) { M = [i]; break; } }
  if (!M.length) return T_('📊 Mois introuvable dans Sales & FP Needs.', '📊 Month not found in Sales & FP Needs.');
  const c = M[0], c2 = M.length > 1 ? M[1] : null;
  const stop = new Set(['sales', 'sale', 'vente', 'ventes', 'vendu', 'vendus', 'sold', 'selling', 'chiffre', 'chiffres', 'turnover', 'volume', 'volumes', 'give', 'details', 'detail', 'detaille', 'pour', 'for', 'the', 'des', 'les', 'month', 'mois', 'versus', 'compare', 'comparaison', 'me', 'donne', 'moi', 'show', 'total', 'par', 'by', 'of', 'de', 'du', 'en', 'in', 'and', 'et', 'vs']);
  const T = waN_(raw).split(' ').filter(w => w.length >= 3 && !stop.has(w) && WA_MON[w] == null && !/^\d+$/.test(w)).map(w => WA_SYN[w] || w);
  let R = t.r.filter(r => r[2] && !/total|max/i.test(r[0] + ' ' + r[2]));
  if (T.length) { const F = R.filter(r => { const W = waN_(r.slice(0, 3).join(' ')).split(' '); return T.every(x => W.some(w => w.indexOf(x) === 0)); }); if (F.length) R = F; }
  const fam = {}; R.forEach(r => { const f = r[0] || T_('Autre', 'Other'), o = fam[f] || (fam[f] = { v: 0, v2: 0 }); o.v += waNum2_(r[c]); if (c2 != null) o.v2 += waNum2_(r[c2]); });
  const pct = (a, b) => b ? ' (' + (a >= b ? '+' : '') + Math.round((a - b) / b * 100) + '%)' : '';
  const f0 = n => Math.round(n).toLocaleString('en-US');
  const L = ['📊 *' + T_('Ventes ', 'Sales ') + H[c] + '*' + (c2 != null ? ' vs ' + H[c2] : '') + T_(' (unités)', ' (units)')];
  Object.keys(fam).sort((a, b) => fam[b].v - fam[a].v).forEach(f => L.push('• *' + f + '* : ' + f0(fam[f].v) + (c2 != null ? ' vs ' + f0(fam[f].v2) + pct(fam[f].v, fam[f].v2) : '')));
  const P = R.map(r => ({ n: r[2], v: waNum2_(r[c]), v2: c2 != null ? waNum2_(r[c2]) : 0 })).filter(p => p.v || p.v2).sort((a, b) => b.v - a.v);
  if (P.length > 1 || T.length) { L.push('', T_('*Par produit*', '*By product*'));
    P.slice(0, 25).forEach(p => L.push('• ' + p.n + ' : ' + f0(p.v) + (c2 != null ? ' vs ' + f0(p.v2) + pct(p.v, p.v2) : '')));
    if (P.length > 25) L.push(T_('… et ' + (P.length - 25) + ' autres', '… and ' + (P.length - 25) + ' more')); }
  return L.join('\n'); }

// Project completion, same rules as the app (PPM): ITE note = 100%, ETA passed = 100%, shipped 30→100% as the ETA nears, ready 30%, in production 15%.
function waPjLines_(all) { const t = waToday_(), out = [];
  (all.ppm || []).forEach(r => { if (r[2] !== 'Projects' || r[13] === 'Hide' || /mohamad ?moussa/i.test(r[0])) return;
    const note = String(r[14] || ''), R = (r[15] || []).map(waPd_).filter(Boolean), E = (r[16] || []).map(waPd_).filter(Boolean), A = (r[17] || []).map(waPd_).filter(Boolean);
    const arrTxt = /arriv/i.test((r[17] || []).join(' ')), eta = A.length ? new Date(Math.max.apply(null, A)) : null, etd = E.length ? new Date(Math.min.apply(null, E)) : null;
    let dp = 0, st = T_('Pas démarré', 'Not started');
    if (/(^|\W)ite(\W|$)/i.test(note)) { dp = 100; st = T_('À ITE', 'At ITE'); }
    else if ((eta && eta <= t) || arrTxt) { dp = 100; st = T_('Arrivé', 'Arrived'); }
    else if ((etd && etd <= t) || (eta && !etd)) { const s0 = etd || new Date(+eta - 60 * 864e5), f = eta ? Math.min(1, Math.max(0, (t - s0) / (eta - s0))) : 0; dp = Math.round(30 + 70 * f); st = T_('Expédié', 'Shipped'); }
    else if (E.length || R.some(d => d <= t)) { dp = 30; st = T_('Prêt à expédier', 'Ready to ship'); }
    else if (R.length) { dp = 15; st = T_('En production', 'In production'); }
    out.push({ sup: r[0], item: r[1], fam: r[3] || T_('Autre', 'Other'), v: Math.abs((r[5] || 0) * waFx_(all, r[4])) || 1, dp: dp, st: st, eta: eta }); });
  return out; }
function waProjects_(all, raw) { const P = waPjLines_(all); if (!P.length) return T_('🏗️ Aucun projet.', '🏗️ No project.');
  const q = waN_(raw), fams = [...new Set(P.map(p => p.fam))], pick = fams.filter(f => (' ' + q + ' ').indexOf(' ' + waN_(f) + ' ') >= 0);
  const wav = L => Math.round(L.reduce((s, p) => s + p.v * p.dp, 0) / L.reduce((s, p) => s + p.v, 0));
  if (!pick.length) { const L = ['🏗️ *' + T_('Avancement des projets', 'Project completion') + '* : ' + wav(P) + '%'];
    fams.map(f => [f, P.filter(p => p.fam === f)]).sort((a, b) => wav(a[1]) - wav(b[1])).forEach(x => L.push('• *' + x[0] + '* : ' + wav(x[1]) + '% · ' + x[1].length + T_(' ligne(s)', ' line(s)')));
    L.push('', T_('_Envoyez « projets water » pour le détail, ou « master plan pdf »._', '_Send “projects water” for details, or “master plan pdf”._')); return L.join('\n'); }
  return pick.map(f => { const L = P.filter(p => p.fam === f).sort((a, b) => a.dp - b.dp);
    return ['🏗️ *' + f + '* : ' + wav(L) + '%'].concat(L.map(p => '• ' + p.sup + ' – ' + p.item + ' : ' + p.dp + '% · ' + p.st + (p.eta ? ' · ETA ' + waDay_(p.eta) : ''))).join('\n'); }).join('\n\n'); }
